Database Schema
[NEW] returns table
Tracks the high-level return request from a user.

id (UUID, PK)
order_id (UUID, FK -> orders)
user_id (UUID, FK -> users)
status (Enum: 'pending', 'approved', 'rejected', 'refunded')
refund_amount (Numeric)
reason (Text)
admin_notes (Text)
created_at, updated_at
[NEW] return_items table
Tracks specific items within a return request.

id (UUID, PK)
return_id (UUID, FK -> returns)
medicine_id (UUID, FK -> medicines)
quantity (Integer)
condition (Enum: 'good', 'damaged', 'expired')
[NEW] inventory_adjustments table
Immutable ledger of all stock changes.

id (UUID, PK)
medicine_id (UUID, FK -> medicines)
adjustment_type (Enum: 'restock', 'sale', 'return_restock', 'return_discard', 'damage', 'expired', 'correction', 'audit')
quantity_change (Integer) - Positive for additions, negative for reductions.
reference_id (UUID) - ID of the order or return causing the change (nullable).
performed_by (UUID, FK -> users)
reason (Text)
created_at
Backend Logic
[NEW] src/controllers/returnController.js
requestReturn(orderId, items): User initiates a return.
approveReturn(returnId): Admin approves. Triggers inventory adjustment and refund logic.
rejectReturn(returnId): Admin rejects.
[NEW] src/controllers/inventoryController.js
logAdjustment(medicineId, type, quantity, user, reason): Central function to update medicines.quantity and insert into inventory_adjustments transactionally.
[MODIFY] src/controllers/orderController.js
Update createOrder to log 'sale' adjustments in inventory_adjustments.

[MODIFY] view/orders/index.ejs (or equivalent)
Add "Request Return" button for eligible orders.
[NEW] view/admin/returns.ejs
Admin interface to view and process returns.

                                if (currentValue !== String(value)) {
                                    el.style.transform = 'scale(1.2)';
                                    setTimeout(() => {
                                        el.textContent = id === 'counter-revenue' ? `$${value}` : value;
                                        el.style.transform = 'scale(1)';
                                    }, 150);
                                }
                            }
                        };

                        // Update all counters
                        updateCounter('counter-revenue', data.data.revenue);
                        updateCounter('counter-orders', data.data.ordersCount);
                        updateCounter('counter-prescriptions', data.data.prescriptionsCount);
                        updateCounter('counter-total-requests', data.data.totalRequests);

                        // Update alerts
                        const updateAlert = (id, value) => {
                            const el = document.getElementById(id);
                            if (el) {
                                if (el.textContent !== String(value)) {
                                    el.classList.add('animate-pulse');
                                    el.textContent = value;
                                    setTimeout(() => el.classList.remove('animate-pulse'), 1000);
                                }
                            }
                        };

                        updateAlert('alert-low-stock-count', data.data.lowStockCount);
                        updateAlert('alert-scheduled-count', data.data.scheduledOrdersCount);
                    }
                } catch (err) {
                    console.error('Dashboard stats update failed:', err);
                }
            }

                let max = parseInt(input.getAttribute('max'));
                let newVal = current + delta;

                if (newVal >= 1 && newVal <= max) {
                    input.value = newVal;
                    display.innerText = newVal;
                    calculateReturnTotal();
                }
            }

                calculateReturnTotal();

            function calculateReturnTotal() {
                let total = 0;
                const checkboxes = document.querySelectorAll('#return-items-list input[type="checkbox"]:checked');

                checkboxes.forEach(cb => {
                    const itemId = cb.value;
                    const item = returnItemsCache.find(i => i.id === itemId);
                    const qty = parseInt(document.getElementById(`qty-${itemId}`).value);
                    if (item) {
                        total += parseFloat(item.price) * qty;
                    }
                });

                document.getElementById('return-total-refund').textContent = `$${total.toFixed(2)}`;
            }

            async function submitReturn() {
                if (!selectedReturnOrderId) return;

                const checkboxes = document.querySelectorAll('#return-items-list input[type="checkbox"]:checked');
                if (checkboxes.length === 0) return showToast('error', 'Select at least one item');

                if (!confirm("Confirm Return? \nThis will deduct revenue and update stock (if selected).")) return;

                const selectedItems = [];
                checkboxes.forEach(cb => {
                    const itemId = cb.value;
                    const item = returnItemsCache.find(i => i.id === itemId);
                    const qty = parseInt(document.getElementById(`qty-${itemId}`).value);
                    selectedItems.push({
                        medicine_id: item.medicine_id,
                        quantity: qty,
                        price: item.price
                    });
                });

                const payload = {
                    orderId: selectedReturnOrderId,
                    items: selectedItems,
                    reason: document.getElementById('return-reason').value,
                    condition: document.getElementById('return-condition').value,
                    updateInventory: document.getElementById('return-inventory-update').checked,
                    notes: document.getElementById('return-notes').value
                };

                try {
                    const res = await fetch('/doctor/returns/process', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });
                    const data = await res.json();

                    if (data.success) {
                        showToast('success', 'Return Processed Successfully');
                        closeReturnModal();
                        loadRecentReturns();
                    } else {
                        showToast('error', data.error || 'Processing Failed');
                    }
                } catch (e) {
                    showToast('error', 'Network Error');
                }
            }

            let currentOrderStatus = 'pending';

            async function loadOrders() {
                try {
                    const res = await fetch('/doctor/orders/all');
                    const data = await res.json();

                    if (data.success) {
                        ordersCache = data.orders;
                        updateOrderCounts(data.orders);
                        filterOrdersByStatus(currentOrderStatus, false);
                    }

                    console.error('Failed to load orders:', err);
                }

            function updateOrderCounts(orders) {
                const counts = {
                    pending: 0,
                    scheduled: 0,
                    processing: 0,
                    completed: 0,
                    canceled: 0
                };

                orders.forEach(order => {
                    if (counts.hasOwnProperty(order.status)) {
                        counts[order.status]++;
                    }
                });

                Object.keys(counts).forEach(status => {
                    const el = document.getElementById(`count-${status}`);
                    if (el) el.textContent = counts[status];
                });
            }

            function filterOrdersByStatus(status, shouldLoad = true) {
                currentOrderStatus = status;

                // Update tab active state
                document.querySelectorAll('.order-status-tab').forEach(tab => {
                    tab.classList.remove('active', 'bg-primary', 'text-white');
                    tab.classList.add('text-gray-600', 'dark:text-gray-400');
                });
                const activeTab = document.querySelector(`[data-status="${status}"]`);
                if (activeTab) {
                    activeTab.classList.add('active', 'bg-primary', 'text-white');
                    activeTab.classList.remove('text-gray-600', 'dark:text-gray-400');
                }

                if (shouldLoad) {
                    loadOrders();
                } else {
                    renderOrders(ordersCache.filter(o => o.status === status));
                }
            }

            function toggleOrderDetails(orderId) {
                const details = document.getElementById(`details-${orderId}`);
                const chevron = document.getElementById(`chevron-${orderId}`);
                const itemsContainer = document.getElementById(`items-${orderId}`);

                if (details.classList.contains('hidden')) {
                    details.classList.remove('hidden');
                    chevron.style.transform = 'rotate(180deg)';

                    // Load order items if not loaded yet
                    if (itemsContainer.innerHTML.includes('Loading')) {
                        loadOrderItems(orderId);
                    }
                } else {
                    details.classList.add('hidden');
                    chevron.style.transform = 'rotate(0deg)';
                }
            }

            async function changeOrderStatus(orderId, newStatus) {
                if (!confirm(`Are you sure you want to change this order to ${newStatus}?`)) return;

                try {
                    const res = await fetch('/doctor/orders/status', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({ orderId, newState: newStatus })
                    });

                    if (res.ok) {
                        showToast('success', `Order ${newStatus}!`);
                        loadOrders(); // Reload to update
                    } else {
                        showToast('error', 'Failed to update order');
                    }
                } catch (err) {
                    console.error(err);
                    showToast('error', 'Failed to update order');
                }
            }
